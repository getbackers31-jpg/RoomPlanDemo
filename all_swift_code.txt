private extension AnnotationsDetailsView {
    func closeView() {
        selectedAnnotation?.removeHighlight()
        selectedAnnotation = nil
    }
    
    func editAnnotation(isEditingTitleLabel: Bool) {
        if isEditingTitleLabel {
            newTitle = selectedAnnotation?.titleLabel.text ?? ""
            isEditingTitle = true
        } else {
            newTitle = selectedAnnotation?.subtitleLabel.text ?? ""
            isEditingSubTitle = true
        }
    }
    
    func action(for type: DetailsType) {
        switch type {
        case .delete:
            selectedAnnotation?.removeFromParent(); NotificationCenter.default.post(name: .floorPlanModelDidMutate, object: "delete_annotation")
            selectedAnnotation = nil
        case .rotate:
            let normalizedAngle = (angle + 15).truncatingRemainder(dividingBy: 360)
            selectedAnnotation?.updateAngle(to: normalizedAngle)
            angle = normalizedAngle
            NotificationCenter.default.post(name: .floorPlanModelDidMutate, object: "rotate_annotation")
        default: break
        }
    }
    
    func updateColor(_ color: MainColor) {
        selectedAnnotation?.updateColor(color.skColor)
        NotificationCenter.default.post(name: .floorPlanModelDidMutate, object: "edit_annotation_color")
    }
}

import SwiftUI
import RoomPlan

struct ObjectDetailsView: View {
    @Binding var object: FloorPlanObject?
    @Binding var selectedType: DetailsType?
    
    @State private var angle: CGFloat = 0
    
    init(object: Binding<FloorPlanObject?>,
         selectedType: Binding<DetailsType?>) {
        _object = object
        _selectedType = selectedType
    }
    
    var body: some View {
        VStack(alignment: .leading, spacing: 10) {
            HStack {
                VStack(alignment: .leading, spacing: 20) {
                    headerView
                    
                    ScrollView(.horizontal) {
                        HStack(spacing: 20) {
                            ForEach(DetailsType.allCases, id: \.self) { type in
                                createSliderButton(for: type)
                            }
                        }
                    }
                }
                .padding(20)
            }
            
            Spacer()
        }
        .background(Color.teal)
        .clipShape(RoundedRectangle(cornerRadius: 12))
        .onAppear {
            if let object {
                // Convert radians to degrees
                let rotationInDegrees = object.zRotation * (180 / .pi)
                
                // Snap the angle to the nearest 15 degrees
                let snappedAngle = round(rotationInDegrees / 15) * 15
                
                // Normalize the angle to keep it within 0??60 degrees
                let normalizedAngle = snappedAngle.truncatingRemainder(dividingBy: 360)
                
                angle = normalizedAngle
            }
        }
    }
}

                            .foregroundStyle(.white)

                }
            case .rotate:

                        object.updateAngle(to: normalizedAngle)
                        angle = normalizedAngle
                        NotificationCenter.default.post(name: .floorPlanModelDidMutate, object: "rotate_object")

import CoreGraphics

extension CGPoint {
    
    func rotateAround(point: CGPoint, by angle: CGFloat) -> CGPoint {
        // Translate to origin
        let x1 = self.x - point.x
        let y1 = self.y - point.y
        
        // Apply rotation
        let x2 = x1 * cos(angle) - y1 * sin(angle)
        let y2 = x1 * sin(angle) + y1 * cos(angle)
        
        // Translate back
        let newX = x2 + point.x
        let newY = y2 + point.y
        
        return CGPoint(x: newX, y: newY)
    }
    
}

import RoomPlan
import SceneKit
import SwiftUI
import UIKit

struct FloorPlan3DPreviewView: UIViewRepresentable {
    let capturedRoom: CapturedRoom
    let editState: FloorPlanEditState

    func makeUIView(context: Context) -> SCNView {
        let scnView = SCNView()
        scnView.scene = makeScene()
        scnView.allowsCameraControl = true
        scnView.autoenablesDefaultLighting = false
        scnView.backgroundColor = UIColor(red: 0.91, green: 0.93, blue: 0.96, alpha: 1)
        scnView.antialiasingMode = .multisampling4X
        return scnView
    }

    func updateUIView(_ scnView: SCNView, context: Context) {}

    private func makeScene() -> SCNScene {
        let scene = SCNScene()
        let rootNode = scene.rootNode
        let scale = CGFloat(DrawParameters.scalingFactor)
        let geometry = FloorPlanGeometry(
            capturedRoom: capturedRoom,
            editState: editState
        )

        var minX = Float.greatestFiniteMagnitude
        var maxX = -Float.greatestFiniteMagnitude
        var minZ = Float.greatestFiniteMagnitude
        var maxZ = -Float.greatestFiniteMagnitude
        var maxHeight: Float = 2.6

        func includeBounds(x: Float, z: Float, radius: Float) {
            minX = min(minX, x - radius)
            maxX = max(maxX, x + radius)
            minZ = min(minZ, z - radius)
            maxZ = max(maxZ, z + radius)
        }

        for wall in geometry.joinedWallSegments {
            let width = max(wall.length / scale, 0.12)
            let height = max(wall.heightInMeters, 0.1)
            let depth = max(wall.depthInMeters, 0.08)
            let center = CGPoint(
                x: -wall.midpoint.x / scale,
                y: -wall.midpoint.y / scale
            )
            let yaw = -wall.angle
            let box = SCNBox(width: width, height: height, length: depth, chamferRadius: 0)
            box.firstMaterial = makeMaterial(
                color: UIColor(red: 0.76, green: 0.79, blue: 0.84, alpha: 1),
                transparency: 0.52
            )
            let node = SCNNode(geometry: box)
            node.position = SCNVector3(Float(center.x), Float(height / 2), Float(center.y))
            node.eulerAngles = SCNVector3(0, Float(yaw), 0)
            node.name = "wall"
            rootNode.addChildNode(node)

            includeBounds(
                x: Float(center.x),
                z: Float(center.y),
                radius: Float(width / 2 + depth)
            )
            maxHeight = max(maxHeight, Float(height))
        }

        for surface in geometry.surfaces where !surface.isWallGeometry {
            let representation = surface.effectiveRepresentation
            if representation == .wideOpening {
                continue
            }
            if representation == .beam || representation == .column {
                let width = max(surface.widthInMeters, 0.12)
                let depth = max(surface.depthInMeters, 0.08)
                let height = representation == .beam
                    ? max(surface.heightInMeters * 0.12, 0.2)
                    : surface.heightInMeters
                let worldCenter = surface.worldCenter
                let centerY = representation == .beam
                    ? max(surface.heightInMeters - height / 2, height / 2)
                    : height / 2
                let box = SCNBox(width: width, height: height, length: depth, chamferRadius: 0)
                box.firstMaterial = makeMaterial(color: UIColor(red: 0.47, green: 0.31, blue: 0.72, alpha: 1))
                let node = SCNNode(geometry: box)
                node.position = SCNVector3(Float(worldCenter.x), Float(centerY), Float(worldCenter.y))
                node.eulerAngles = SCNVector3(0, Float(-surface.angle), 0)
                node.name = representation == .beam ? "beamMarker" : "columnMarker"
                rootNode.addChildNode(node)
                includeBounds(
                    x: Float(worldCenter.x),
                    z: Float(worldCenter.y),
                    radius: Float(width / 2 + depth)
                )
                maxHeight = max(maxHeight, Float(centerY + height / 2))
                continue
            }

            let isDoor: Bool
            let isOpening: Bool
            let color: UIColor
            let transparency: CGFloat
            switch surface.source.category {
            case .wall:
                continue
            case .door:
                isDoor = true
                isOpening = false
                color = UIColor(red: 0.68, green: 0.39, blue: 0.22, alpha: 1)
                transparency = 1
            case .opening:
                isDoor = false
                isOpening = true
                color = UIColor(red: 0.38, green: 0.72, blue: 0.69, alpha: 1)
                transparency = 0.24
            case .window:
                isDoor = false
                isOpening = false
                color = UIColor(red: 0.22, green: 0.57, blue: 0.86, alpha: 1)
                transparency = surface.isFloorToCeilingGlass ? 0.45 : 0.72
            default:
                continue
            }

            let centerX = Float(surface.worldCenter.x)
            let centerZ = Float(surface.worldCenter.y)
            let yaw = Float(-surface.angle)
            let width = max(surface.widthInMeters, 0.1)
            let height = surface.heightInMeters
            let depth = max(surface.depthInMeters, 0.08)
            let centerY = max(height / 2, surface.centerHeightInMeters)

import Foundation

enum FloorPlanSurfaceRepresentation: String, Codable, Equatable {
    case source
    case solidWall
    case wideOpening
    case beam
    case column
}

struct FloorPlanSurfaceGeometry {
    let source: CapturedRoom.Surface
    let center: CGPoint
    let angle: CGFloat
    let halfLength: CGFloat
    let representation: FloorPlanSurfaceRepresentation
    let isFloorToCeilingGlass: Bool
    let openDirection: OpenDirection
    let strokeWidth: CGFloat?
    let scaleFactor: CGFloat
    let pointsPerMeter: CGFloat

    var widthInMeters: CGFloat {
        let measuredWidth = halfLength * 2 / pointsPerMeter
        if representation == .column {
            return min(max(measuredWidth, 0.12), 0.2)
        }
        return measuredWidth
    }

    var heightInMeters: CGFloat {
        let measured = CGFloat(source.dimensions.y)
        let fallback: CGFloat
        if source.category == .wall || effectiveRepresentation == .solidWall {
            fallback = 2.6
        } else if source.category == .door || source.category == .opening {
            fallback = 2.05
        } else {
            fallback = 1.2
        }
        return max((measured.isFinite && measured > 0.1 ? measured : fallback) * scaleFactor, 0.1)
    }

    var depthInMeters: CGFloat {
        max(CGFloat(source.dimensions.z) * scaleFactor, 0.06)
    }

    var worldCenter: CGPoint {
        CGPoint(x: -center.x / pointsPerMeter, y: -center.y / pointsPerMeter)
    }

    var centerHeightInMeters: CGFloat {
        CGFloat(source.transform.position.y) * scaleFactor
    }

    var effectiveRepresentation: FloorPlanSurfaceRepresentation {
        if representation == .source,
           source.category == .door,
           halfLength * 2 / pointsPerMeter > oversizedDoorWidthMeters {
            return .wideOpening
        }
        return representation
    }

    var isWallGeometry: Bool {
        switch effectiveRepresentation {
        case .solidWall:
            return true
        case .source:
            return source.category == .wall
        case .wideOpening, .beam, .column:
            return false
        }
    }

    var isNoiseFragment: Bool {
        isWallGeometry && widthInMeters < FloorPlanGeometry.minimumWallSegmentLengthMeters
    }
}

    let baseDepthInPoints: CGFloat

    let pointsPerMeter: CGFloat

struct FloorPlanWallSegment {

    var end: CGPoint
    var heightInMeters: CGFloat = 2.6

struct FloorPlanGeometry {
    static let minimumWallSegmentLengthMeters: CGFloat = 0.12
    static let oversizedDoorWidthMeters: CGFloat = 2.2

    let scaleFactor: CGFloat
    let pointsPerMeter: CGFloat
    let surfaces: [FloorPlanSurfaceGeometry]
    let objects: [FloorPlanObjectGeometry]
    let manualObjects: [FloorPlanManualObjectGeometry]

    var joinedWallSegments: [FloorPlanWallSegment] {
        Self.joinedWallSegments(
            from: surfaces.filter(\.isWallGeometry).map { surface in
                Self.makeWallSegment(
                    center: surface.center,
                    angle: surface.angle,
                    halfLength: surface.halfLength,
                    height: surface.heightInMeters,
                    depth: surface.depthInMeters
                )
            },
            pointsPerMeter: pointsPerMeter
        )
    }

    init(

        scaleFactor: CGFloat? = nil,
        pointsPerMeter: CGFloat = DrawParameters.scalingFactor
    ) {
        let calibrationFactor = scaleFactor
            ?? CGFloat(editState?.calibration?.scaleFactor ?? 1)
        let resolvedScaleFactor = calibrationFactor.isFinite && calibrationFactor > 0
            ? calibrationFactor
            : 1
        self.scaleFactor = resolvedScaleFactor
        self.pointsPerMeter = pointsPerMeter

        surfaces = (capturedRoom.doors + capturedRoom.openings + capturedRoom.walls + capturedRoom.windows)
            .map { surface in
                let saved = editState?.surfaces[surface.identifier.uuidString]
                return FloorPlanSurfaceGeometry(

                            ?? Double(-CGFloat(surface.transform.eulerAngles.z - surface.transform.eulerAngles.y))
                    ),
                    halfLength: CGFloat(

                            ?? Double(CGFloat(surface.dimensions.x) * pointsPerMeter * resolvedScaleFactor / 2)
                    ),
                    representation: saved?.representation ?? .source,
                    isFloorToCeilingGlass: saved?.isFloorToCeilingGlass ?? false,
                    openDirection: saved?.openDirection ?? .clockwise,

                    pointsPerMeter: pointsPerMeter
                )

        objects = capturedRoom.objects.map { object in

            let baseWidthScale = saved == nil ? resolvedScaleFactor : 1
            return FloorPlanObjectGeometry(

                ),

                ),

                baseWidthInPoints: max(CGFloat(object.dimensions.x) * pointsPerMeter * baseWidthScale, 40),
                baseDepthInPoints: max(CGFloat(object.dimensions.z) * pointsPerMeter * baseWidthScale, 40),
                scaleFactor: resolvedScaleFactor,

    static func joinedWallSegments(
        from source: [FloorPlanWallSegment],
        pointsPerMeter: CGFloat = DrawParameters.scalingFactor
    ) -> [FloorPlanWallSegment] {
        let minimumLength = pointsPerMeter * minimumWallSegmentLengthMeters
        var segments = source.filter { $0.length >= minimumLength }
        let parallelTolerance = sin(CGFloat.pi / 45)
        let offsetTolerance = DrawParameters.wallWidth * 0.5
        let gapTolerance = DrawParameters.wallWidth * 1.5

        var didMerge = true
        while didMerge {
            didMerge = false
            guard segments.count > 1 else { break }

            mergeSearch: for firstIndex in 0..<(segments.count - 1) {
                for secondIndex in (firstIndex + 1)..<segments.count {
                    let first = segments[firstIndex]
                    let second = segments[secondIndex]
                    let firstVector = CGPoint(x: first.end.x - first.start.x, y: first.end.y - first.start.y)
                    let secondVector = CGPoint(x: second.end.x - second.start.x, y: second.end.y - second.start.y)
                    let firstLength = first.length
                    let secondLength = second.length
                    guard firstLength > 1, secondLength > 1 else { continue }

                    let firstDirection = CGPoint(x: firstVector.x / firstLength, y: firstVector.y / firstLength)
                    let secondDirection = CGPoint(x: secondVector.x / secondLength, y: secondVector.y / secondLength)
                    guard abs(crossProduct(firstDirection, secondDirection)) <= parallelTolerance else { continue }

                    let directionSign: CGFloat = dotProduct(firstDirection, secondDirection) < 0 ? -1 : 1
                    let combinedVector = CGPoint(
                        x: firstVector.x + secondVector.x * directionSign,
                        y: firstVector.y + secondVector.y * directionSign
                    )
                    let combinedLength = hypot(combinedVector.x, combinedVector.y)
                    guard combinedLength > 1 else { continue }
                    let direction = CGPoint(x: combinedVector.x / combinedLength, y: combinedVector.y / combinedLength)
                    let normal = CGPoint(x: -direction.y, y: direction.x)
                    let firstStart = dotProduct(first.start, direction)
                    let firstEnd = dotProduct(first.end, direction)
                    let secondStart = dotProduct(second.start, direction)
                    let secondEnd = dotProduct(second.end, direction)
                    let firstMin = min(firstStart, firstEnd)
                    let firstMax = max(firstStart, firstEnd)
                    let secondMin = min(secondStart, secondEnd)
                    let secondMax = max(secondStart, secondEnd)
                    let gap = max(0, max(firstMin, secondMin) - min(firstMax, secondMax))
                    guard gap <= gapTolerance else { continue }

                    let overlapStart = max(firstMin, secondMin)
                    let overlapEnd = min(firstMax, secondMax)
                    let comparisonProjection = overlapStart <= overlapEnd
                        ? (overlapStart + overlapEnd) / 2
                        : (max(firstMin, secondMin) + min(firstMax, secondMax)) / 2
                    let firstStartOffset = dotProduct(first.start, normal)
                    let firstEndOffset = dotProduct(first.end, normal)
                    let secondStartOffset = dotProduct(second.start, normal)
                    let secondEndOffset = dotProduct(second.end, normal)
                    let firstOffsetAtComparison = firstStartOffset
                        + (comparisonProjection - firstStart) * (firstEndOffset - firstStartOffset) / (firstEnd - firstStart)
                    let secondOffsetAtComparison = secondStartOffset
                        + (comparisonProjection - secondStart) * (secondEndOffset - secondStartOffset) / (secondEnd - secondStart)
                    guard abs(firstOffsetAtComparison - secondOffsetAtComparison) <= offsetTolerance else { continue }

                    let firstMidOffset = (firstStartOffset + firstEndOffset) / 2
                    let secondMidOffset = (secondStartOffset + secondEndOffset) / 2
                    let averageOffset = (firstMidOffset * firstLength + secondMidOffset * secondLength)
                        / (firstLength + secondLength)
                    segments[firstIndex] = FloorPlanWallSegment(
                        start: CGPoint(
                            x: direction.x * min(firstMin, secondMin) + normal.x * averageOffset,
                            y: direction.y * min(firstMin, secondMin) + normal.y * averageOffset
                        ),
                        end: CGPoint(
                            x: direction.x * max(firstMax, secondMax) + normal.x * averageOffset,
                            y: direction.y * max(firstMax, secondMax) + normal.y * averageOffset
                        ),
                        heightInMeters: max(first.heightInMeters, second.heightInMeters),
                        depthInMeters: max(first.depthInMeters, second.depthInMeters)
                    )
                    segments.remove(at: secondIndex)
                    didMerge = true
                    break mergeSearch
                }
            }
        }

        return connectCorners(segments)
    }

    static func segments(from surfaces: [FloorPlanSurface]) -> [FloorPlanWallSegment] {
        surfaces.map {
            makeWallSegment(
                center: $0.position,
                angle: $0.zRotation,
                halfLength: $0.halfLength,
                height: CGFloat($0.capturedSurface.dimensions.y),
                depth: CGFloat($0.capturedSurface.dimensions.z)
            )
        }

    private static func makeWallSegment(
        center: CGPoint,
        angle: CGFloat,
        halfLength: CGFloat,
        height: CGFloat,
        depth: CGFloat
    ) -> FloorPlanWallSegment {
        let direction = CGPoint(x: cos(angle), y: sin(angle))
        return FloorPlanWallSegment(
            start: CGPoint(x: center.x - halfLength * direction.x, y: center.y - halfLength * direction.y),
            end: CGPoint(x: center.x + halfLength * direction.x, y: center.y + halfLength * direction.y),
            heightInMeters: height,
            depthInMeters: depth
        )
    }

    private static func connectCorners(_ source: [FloorPlanWallSegment]) -> [FloorPlanWallSegment] {
        guard source.count > 1 else { return source }

        var connected = source
        let minimumCrossProduct = sin(CGFloat.pi / 45)
        let endpointTolerance = DrawParameters.wallWidth * 2
        let maximumMiterExtension = DrawParameters.wallWidth * 3

        func endpoint(_ isStart: Bool, of segment: FloorPlanWallSegment) -> CGPoint {
            isStart ? segment.start : segment.end
        }

        for firstIndex in 0..<(source.count - 1) {
            for secondIndex in (firstIndex + 1)..<source.count {
                let first = source[firstIndex]
                let second = source[secondIndex]
                let firstVector = CGPoint(x: first.end.x - first.start.x, y: first.end.y - first.start.y)
                let secondVector = CGPoint(x: second.end.x - second.start.x, y: second.end.y - second.start.y)
                let firstLength = first.length
                let secondLength = second.length
                guard firstLength > 1, secondLength > 1 else { continue }

                let firstDirection = CGPoint(x: firstVector.x / firstLength, y: firstVector.y / firstLength)
                let secondDirection = CGPoint(x: secondVector.x / secondLength, y: secondVector.y / secondLength)
                let determinant = crossProduct(firstDirection, secondDirection)
                guard abs(determinant) > minimumCrossProduct else { continue }

                var bestMatch: (point: CGPoint, firstIsStart: Bool, secondIsStart: Bool, score: CGFloat)?
                for firstIsStart in [true, false] {
                    let firstPoint = endpoint(firstIsStart, of: first)
                    for secondIsStart in [true, false] {
                        let secondPoint = endpoint(secondIsStart, of: second)
                        let endpointDistance = hypot(firstPoint.x - secondPoint.x, firstPoint.y - secondPoint.y)
                        guard endpointDistance <= endpointTolerance else { continue }

                        let between = CGPoint(x: secondPoint.x - firstPoint.x, y: secondPoint.y - firstPoint.y)
                        let distanceAlongFirst = crossProduct(between, secondDirection) / determinant
                        let intersection = CGPoint(
                            x: firstPoint.x + firstDirection.x * distanceAlongFirst,
                            y: firstPoint.y + firstDirection.y * distanceAlongFirst
                        )
                        let firstExtension = hypot(intersection.x - firstPoint.x, intersection.y - firstPoint.y)
                        let secondExtension = hypot(intersection.x - secondPoint.x, intersection.y - secondPoint.y)
                        guard firstExtension <= maximumMiterExtension,
                              secondExtension <= maximumMiterExtension else { continue }

                        let score = endpointDistance + firstExtension + secondExtension
                        if let current = bestMatch, current.score <= score { continue }
                        bestMatch = (intersection, firstIsStart, secondIsStart, score)
                    }
                }

                guard let match = bestMatch else { continue }
                if match.firstIsStart {
                    connected[firstIndex].start = match.point
                } else {
                    connected[firstIndex].end = match.point
                }
                if match.secondIsStart {
                    connected[secondIndex].start = match.point
                } else {
                    connected[secondIndex].end = match.point
                }
            }
        }

        return connected
    }

    private static func crossProduct(_ lhs: CGPoint, _ rhs: CGPoint) -> CGFloat {
        lhs.x * rhs.y - lhs.y * rhs.x
    }

    private static func dotProduct(_ lhs: CGPoint, _ rhs: CGPoint) -> CGFloat {
        lhs.x * rhs.x + lhs.y * rhs.y
    }
}

        var coord = -extent

            target.move(to: CGPoint(x: coord, y: -extent))
            target.addLine(to: CGPoint(x: coord, y: extent))

    private func computePolygonAreaPt2(from segments: [(CGPoint, CGPoint)]) -> Double {
        let tolerance: CGFloat = 40.0
        var vertices: [CGPoint] = []
        
        func vertexIndex(for pt: CGPoint) -> Int {
            for (idx, existing) in vertices.enumerated() {
                if hypot(existing.x - pt.x, existing.y - pt.y) <= tolerance {
                    return idx
                }
            }
            vertices.append(pt)
            return vertices.count - 1
        }
        
        var adjacency: [Int: Set<Int>] = [:]
        for (p1, p2) in segments {
            let u = vertexIndex(for: p1)
            let v = vertexIndex(for: p2)
            if u == v { continue }
            adjacency[u, default: []].insert(v)
            adjacency[v, default: []].insert(u)
        }
        
        func shoelace(_ pts: [CGPoint]) -> Double {
            guard pts.count >= 3 else { return 0.0 }
            var sum: Double = 0.0
            for i in 0..<pts.count {
                let j = (i + 1) % pts.count
                sum += Double(pts[i].x * pts[j].y - pts[j].x * pts[i].y)
            }
            return abs(sum) * 0.5
        }
        
        var visitedDirectedEdges: Set<Int> = []
        func edgeKey(_ u: Int, _ v: Int) -> Int { u * 10000 + v }
        
        var bestCycleArea: Double = 0.0
        var totalLoops = 0
        for (u, neighbors) in adjacency {
            for v in neighbors {
                if totalLoops > 2000 { break }
                totalLoops += 1
                
                if visitedDirectedEdges.contains(edgeKey(u, v)) { continue }
                var path: [Int] = [u]
                var currU = u
                var currV = v
                var steps = 0
                
                while steps < 100 {
                    visitedDirectedEdges.insert(edgeKey(currU, currV))
                    if currV == u {
                        if path.count >= 3 {
                            let area = shoelace(path.map { vertices[$0] })
                            if area > bestCycleArea { bestCycleArea = area }
                        }
                        break
                    }
                    if path.contains(currV) { break }
                    path.append(currV)
                    
                    let nextCandidates = (adjacency[currV] ?? []).filter { $0 != currU }
                    guard !nextCandidates.isEmpty else { break }
                    
                    let inAngle = atan2(vertices[currU].y - vertices[currV].y, vertices[currU].x - vertices[currV].x)
                    let nextW = nextCandidates.min { a, b in
                        let angA = atan2(vertices[a].y - vertices[currV].y, vertices[a].x - vertices[currV].x)
                        let angB = atan2(vertices[b].y - vertices[currV].y, vertices[b].x - vertices[currV].x)
                        var diffA = (angA - inAngle).truncatingRemainder(dividingBy: 2 * .pi)
                        if diffA <= 1e-5 { diffA += 2 * .pi }
                        var diffB = (angB - inAngle).truncatingRemainder(dividingBy: 2 * .pi)
                        if diffB <= 1e-5 { diffB += 2 * .pi }
                        return diffA < diffB
                    }!
                    
                    currU = currV
                    currV = nextW
                    steps += 1
                }
            }
        }
        
        if bestCycleArea > 0 {
            return bestCycleArea
        }
        
        guard vertices.count >= 3 else { return 0.0 }
        let cx = vertices.reduce(CGFloat(0)) { $0 + $1.x } / CGFloat(vertices.count)
        let cy = vertices.reduce(CGFloat(0)) { $0 + $1.y } / CGFloat(vertices.count)
        let sorted = vertices.sorted {
            atan2($0.y - cy, $0.x - cx) < atan2($1.y - cy, $1.x - cx)
        }
        return shoelace(sorted)
    }

    private func rotateAllNodes(by deltaAngle: CGFloat) {
        let surfaceNodes = children.compactMap { $0 as? FloorPlanSurface }
        guard !surfaceNodes.isEmpty else { return }
        
        let cx = surfaceNodes.reduce(CGFloat(0)) { $0 + $1.position.x } / CGFloat(surfaceNodes.count)
        let cy = surfaceNodes.reduce(CGFloat(0)) { $0 + $1.position.y } / CGFloat(surfaceNodes.count)
        let cosA = cos(deltaAngle)
        let sinA = sin(deltaAngle)
        
        func rotatePoint(_ pt: CGPoint) -> CGPoint {
            let dx = pt.x - cx
            let dy = pt.y - cy
            return CGPoint(x: cx + dx * cosA - dy * sinA, y: cy + dx * sinA + dy * cosA)
        }
        
        for node in surfaceNodes {
            node.position = rotatePoint(node.position)
            node.zRotation += deltaAngle
            node.updateDimensionLabel()
        }
        
        for obj in children.compactMap({ $0 as? FloorPlanObject }) {
            obj.position = rotatePoint(obj.position)
            obj.zRotation += deltaAngle
            obj.updateSizeLabel()
        }
        
        for note in children.compactMap({ $0 as? FloorPlanAnnotation }) {
            note.position = rotatePoint(note.position)
        }

    private func roomBoundingRect(padding: CGFloat) -> CGRect? {
        let surfaceNodes = children.compactMap { $0 as? FloorPlanSurface }
        guard !surfaceNodes.isEmpty else { return nil }
        
        var minX = CGFloat.greatestFiniteMagnitude
        var maxX = -CGFloat.greatestFiniteMagnitude
        var minY = CGFloat.greatestFiniteMagnitude
        var maxY = -CGFloat.greatestFiniteMagnitude
        
        for node in surfaceNodes {
            let cosA = cos(node.zRotation)
            let sinA = sin(node.zRotation)
            let pts = [
                CGPoint(x: node.position.x - node.halfLength * cosA, y: node.position.y - node.halfLength * sinA),
                CGPoint(x: node.position.x + node.halfLength * cosA, y: node.position.y + node.halfLength * sinA)
            ]
            for pt in pts {
                minX = min(minX, pt.x); maxX = max(maxX, pt.x)
                minY = min(minY, pt.y); maxY = max(maxY, pt.y)
            }
        }
        return CGRect(
            x: minX - padding,
            y: minY - padding,
            width: (maxX - minX) + padding * 2,
            height: (maxY - minY) + padding * 2
        )
    }
    
    func recenterCamera(animated: Bool = true) {
        guard let camera = self.camera,
              let bounds = roomBoundingRect(padding: 230) else { return }
        
        let center = CGPoint(x: bounds.midX, y: bounds.midY)
        let span = max(bounds.width, bounds.height)
        let targetScale = max(0.35, min(2.6, span / 1100.0))
        
        if animated {
            let move = SKAction.move(to: center, duration: 0.25)
            let zoom = SKAction.scale(to: targetScale, duration: 0.25)
            camera.run(SKAction.group([move, zoom]))
        } else {
            camera.position = center
            camera.setScale(targetScale)
        }
    }
    
    private func captureSnapshot() -> SceneSnapshot {
        var surfaceMap: [ObjectIdentifier: SurfaceSnapshot] = [:]
        for node in children.compactMap({ $0 as? FloorPlanSurface }) {

                position: node.position,
                zRotation: node.zRotation,
                halfLength: node.halfLength,
                openDirection: node.openDirection,
                representation: node.representation,
                isFloorToCeilingGlass: node.isFloorToCeilingGlass,
                calibrationScaleFactor: node.calibrationScaleFactor
            )
        }
        var objectMap: [ObjectIdentifier: ObjectSnapshot] = [:]
        for obj in children.compactMap({ $0 as? FloorPlanObject }) {
            objectMap[ObjectIdentifier(obj)] = ObjectSnapshot(
                position: obj.position,
                zRotation: obj.zRotation,
                xScale: obj.xScale,
                yScale: obj.yScale
            )
        }
        return SceneSnapshot(
            surfaces: surfaceMap,
            objects: objectMap,
            cameraRotation: camera?.zRotation ?? 0,
            calibration: viewModel.calibrationState
        )
    }
    
    func pushUndoState() {
        undoStack.append(captureSnapshot())
        if undoStack.count > 25 { undoStack.removeFirst() }
        redoStack.removeAll()
        syncUndoRedoFlags()
    }
    
    private func restoreSnapshot(_ snapshot: SceneSnapshot) {
        for node in children.compactMap({ $0 as? FloorPlanSurface }) {
            if let state = snapshot.surfaces[ObjectIdentifier(node)] {
                node.position = state.position
                node.zRotation = state.zRotation
                node.openDirection = state.openDirection
                node.halfLength = state.halfLength
                node.representation = state.representation
                node.isFloorToCeilingGlass = state.isFloorToCeilingGlass
                node.restoreCalibrationScaleFactor(to: state.calibrationScaleFactor)
                node.updateDimensionLabel()
            }
        }
        for obj in children.compactMap({ $0 as? FloorPlanObject }) {
            if let state = snapshot.objects[ObjectIdentifier(obj)] {
                obj.position = state.position
                obj.zRotation = state.zRotation
                obj.xScale = state.xScale
                obj.yScale = state.yScale
                obj.updateSizeLabel()
            }
        }
        camera?.zRotation = snapshot.cameraRotation
        viewModel.calibrationState = snapshot.calibration
        applyDisplaySettings(
            showFurniture: viewModel.showFurniture,
            showDimensions: viewModel.showDimensions,
            showFurnitureLabels: viewModel.showFurnitureLabels,
            showStructuralMarkers: viewModel.showStructuralMarkers
        )

    }

private extension FloorPlanScene {
    @objc func panGestureAction(_ sender: UIPanGestureRecognizer) {
        let locationInView = sender.location(in: view)
        let locationInScene = convertPoint(fromView: locationInView)
        
        if let selectedObject = viewModel.selectedObject, selectedObject.contains(locationInScene) || isDraggingObject {
            switch sender.state {
            case .began:
                isDraggingObject = true
                pushUndoState()
            case .changed:
                selectedObject.position = locationInScene
                viewModel.markDirty(reason: "move_object")
            case .ended, .cancelled:
                isDraggingObject = false
            default:
                break
            }
            return
        }
        
        if let annotationNode = viewModel.selectedAnnotation, annotationNode.contains(locationInScene) {
            if sender.state == .changed { annotationNode.position = locationInScene; viewModel.markDirty(reason: "move_annotation") }
            return
        }
        
        guard let camera = self.camera else { return }
        switch sender.state {
        case .began:
            previousCameraPosition = camera.position
        case .changed:
            let translation = sender.translation(in: self.view)
            let rotationAngle = -camera.zRotation
            let rotatedTranslation = CGPoint(
                x: translation.x * cos(rotationAngle) - translation.y * sin(rotationAngle),
                y: translation.x * sin(rotationAngle) + translation.y * cos(rotationAngle)
            )
            let translationScale = camera.xScale
            camera.position = CGPoint(
                x: previousCameraPosition.x - rotatedTranslation.x * translationScale,
                y: previousCameraPosition.y - -rotatedTranslation.y * translationScale
            )
        default: break
        }
    }
    
    @objc func pinchGestureAction(_ sender: UIPinchGestureRecognizer) {
        guard let camera = self.camera else { return }
        if sender.state == .began { previousCameraScale = camera.xScale }
        let clampedScale = max(0.2, min(4.0, previousCameraScale * 1 / sender.scale))
        camera.setScale(clampedScale)
    }
    
    @objc func rotationGestureAction(_ sender: UIRotationGestureRecognizer) {
        guard let camera = self.camera else { return }
        if sender.state == .began || sender.state == .changed {
            camera.zRotation += sender.rotation
            sender.rotation = 0
        }
    }
    
    @objc func handleTap(_ gesture: UITapGestureRecognizer) {
        guard let view = self.view else { return }
        let locationInScene = convertPoint(fromView: gesture.location(in: view))
        let hitNodes = nodes(at: locationInScene)
        
        if let annotationNode = findAncestor(ofType: FloorPlanAnnotation.self, in: hitNodes) {
            deselectSurface(); deselectObject(); selectAnnotation(annotationNode)
        } else if let objectNode = findAncestor(ofType: FloorPlanObject.self, in: hitNodes) {
            deselectSurface(); deselectAnnotation(); selectObject(objectNode)
        } else if let surfaceNode = findAncestor(ofType: FloorPlanSurface.self, in: hitNodes) {
            deselectObject(); deselectAnnotation(); selectSurface(surfaceNode)
        } else {
            deselectSurface(); deselectObject(); deselectAnnotation()
        }
    }
}

    func generateDXF() -> String {
        let ptToCm = 100.0 / Double(DrawParameters.scalingFactor)
        
        func fmt(_ v: Double) -> String { String(format: "%.2f", v) }
        func normalizeDeg(_ deg: Double) -> Double {
            var d = deg.truncatingRemainder(dividingBy: 360.0)
            if d < 0 { d += 360.0 }
            return d
        }
        
        var lines: [String] = []
        func addLine(layer: String, x1: Double, y1: Double, x2: Double, y2: Double) {
            lines.append(contentsOf: [
                "0", "LINE",
                "8", layer,
                "10", fmt(x1), "20", fmt(y1), "30", "0.0",
                "11", fmt(x2), "21", fmt(y2), "31", "0.0"
            ])
        }
        
        func addArc(layer: String, cx: Double, cy: Double, radius: Double, startDeg: Double, endDeg: Double) {
            lines.append(contentsOf: [
                "0", "ARC",
                "8", layer,
                "10", fmt(cx), "20", fmt(cy), "30", "0.0",
                "40", fmt(radius),
                "50", fmt(startDeg),
                "51", fmt(endDeg)
            ])
        }
        
        func addText(layer: String, x: Double, y: Double, height: Double, angleDeg: Double, text: String) {
            lines.append(contentsOf: [
                "0", "TEXT",
                "8", layer,
                "10", fmt(x), "20", fmt(y), "30", "0.0",
                "40", fmt(height),
                "50", fmt(angleDeg),
                "1", text
            ])
        }

        func addWallLineAndDimension(cx: Double, cy: Double, halfLength: Double, angle: Double) {
            let cosA = cos(angle)
            let sinA = sin(angle)
            let ax = cx - halfLength * cosA
            let ay = cy - halfLength * sinA
            let bx = cx + halfLength * cosA
            let by = cy + halfLength * sinA
            addLine(layer: "WALLS", x1: ax, y1: ay, x2: bx, y2: by)

            var angleDeg = (angle * 180.0 / .pi).truncatingRemainder(dividingBy: 360.0)
            if angleDeg > 180 { angleDeg -= 360 }
            if angleDeg < -180 { angleDeg += 360 }
            var textAngle = angleDeg
            var offsetDist = 18.0
            if textAngle > 90 { textAngle -= 180; offsetDist = -18.0 }
            if textAngle < -90 { textAngle += 180; offsetDist = -18.0 }
            addText(
                layer: "DIMENSIONS",
                x: cx - sinA * offsetDist,
                y: cy + cosA * offsetDist,
                height: 12.0,
                angleDeg: normalizeDeg(textAngle),
                text: String(format: "%.1f cm", halfLength * 2)
            )
        }
        
        lines.append(contentsOf: [
            "0", "SECTION",
            "2", "HEADER",
            "9", "$INSUNITS",
            "70", "5",
            "0", "ENDSEC"
        ])
        
        let layerDefs: [(String, String)] = [
            ("WALLS", "7"),
            ("DOORS", "1"),
            ("WINDOWS", "5"),
            ("OPENINGS", "4"),
            ("STRUCTURE", "30"),
            ("DIMENSIONS", "3"),
            ("FURNITURE", "8")
        ]
        lines.append(contentsOf: ["0", "SECTION", "2", "TABLES", "0", "TABLE", "2", "LAYER", "70", "\(layerDefs.count)"])
        for (name, color) in layerDefs {
            lines.append(contentsOf: ["0", "LAYER", "2", name, "70", "0", "62", color, "6", "CONTINUOUS"])
        }
        lines.append(contentsOf: ["0", "ENDTAB", "0", "ENDSEC"])
        
        lines.append(contentsOf: ["0", "SECTION", "2", "ENTITIES"])
        
        let surfaceNodes = children.compactMap { $0 as? FloorPlanSurface }
        for surface in surfaceNodes where !surface.isHidden {
            let cx = Double(surface.position.x) * ptToCm
            let cy = Double(surface.position.y) * ptToCm
            let halfLen = Double(surface.halfLength) * ptToCm
            let angle = Double(surface.zRotation)
            let ax = cx - halfLen * cos(angle)
            let ay = cy - halfLen * sin(angle)
            let bx = cx + halfLen * cos(angle)
            let by = cy + halfLen * sin(angle)

            switch surface.effectiveRepresentation {
            case .solidWall:
                addWallLineAndDimension(cx: cx, cy: cy, halfLength: halfLen, angle: angle)
            case .wideOpening:
                addLine(layer: "OPENINGS", x1: ax, y1: ay, x2: bx, y2: by)
            case .beam, .column:
                let markerHalfLength = surface.effectiveRepresentation == .column
                    ? max(min(halfLen, 10.0), 6.0)
                    : halfLen
                let halfDepth = max(Double(surface.capturedSurface.dimensions.z) * 50.0, 4.0)
                let direction = (cos(angle), sin(angle))
                let normal = (-sin(angle), cos(angle))
                let corners = [
                    (cx - markerHalfLength * direction.0 - halfDepth * normal.0,
                     cy - markerHalfLength * direction.1 - halfDepth * normal.1),
                    (cx + markerHalfLength * direction.0 - halfDepth * normal.0,
                     cy + markerHalfLength * direction.1 - halfDepth * normal.1),
                    (cx + markerHalfLength * direction.0 + halfDepth * normal.0,
                     cy + markerHalfLength * direction.1 + halfDepth * normal.1),
                    (cx - markerHalfLength * direction.0 + halfDepth * normal.0,
                     cy - markerHalfLength * direction.1 + halfDepth * normal.1)
                ]
                for index in 0..<4 {
                    let start = corners[index]
                    let end = corners[(index + 1) % 4]
                    addLine(layer: "STRUCTURE", x1: start.0, y1: start.1, x2: end.0, y2: end.1)
                }
            case .source:
                switch surface.capturedSurface.category {
                case .wall:
                    addWallLineAndDimension(cx: cx, cy: cy, halfLength: halfLen, angle: angle)
                case .door:
                    let openAngle = Double(surface.openDirection.angle)
                    let doorEndX = ax + halfLen * 2 * cos(angle + openAngle)
                    let doorEndY = ay + halfLen * 2 * sin(angle + openAngle)
                    addLine(layer: "DOORS", x1: ax, y1: ay, x2: doorEndX, y2: doorEndY)

                    let zRotDeg = angle * 180.0 / .pi
                    let startDeg = surface.openDirection == .clockwise
                        ? normalizeDeg(zRotDeg - 45.0)
                        : normalizeDeg(zRotDeg)
                    let endDeg = surface.openDirection == .clockwise
                        ? normalizeDeg(zRotDeg)
                        : normalizeDeg(zRotDeg + 45.0)
                    addArc(layer: "DOORS", cx: ax, cy: ay, radius: halfLen * 2, startDeg: startDeg, endDeg: endDeg)
                case .window:
                    addLine(layer: "WINDOWS", x1: ax, y1: ay, x2: bx, y2: by)
                case .opening:
                    addLine(layer: "OPENINGS", x1: ax, y1: ay, x2: bx, y2: by)
                default:
                    break
                }
            }
        }
        
        let objectNodes = children.compactMap { $0 as? FloorPlanObject }
        for obj in objectNodes where !obj.isHidden {
            let cx = Double(obj.position.x) * ptToCm
            let cy = Double(obj.position.y) * ptToCm
            let hw = obj.currentWidthCm / 2.0
            let hh = obj.currentHeightCm / 2.0
            let angle = Double(obj.zRotation)
            let cosA = cos(angle)
            let sinA = sin(angle)
            
            let localCorners = [(-hw, -hh), (hw, -hh), (hw, hh), (-hw, hh)]
            let worldCorners = localCorners.map { (lx, ly) in
                (cx + lx * cosA - ly * sinA, cy + lx * sinA + ly * cosA)
            }
            for i in 0..<4 {
                let p1 = worldCorners[i]
                let p2 = worldCorners[(i + 1) % 4]
                addLine(layer: "FURNITURE", x1: p1.0, y1: p1.1, x2: p2.0, y2: p2.1)
            }
        }
        
        lines.append(contentsOf: ["0", "ENDSEC", "0", "EOF"])
        return lines.joined(separator: "\r\n")
    }

private extension FloorPlanSurface {
    func markDirty(reason: String) {
        (scene as? FloorPlanScene)?.viewModel.markDirty(reason: reason)
    }

    func drawSurface() {
        switch effectiveRepresentation {
        case .solidWall:
            drawWall()
            updateDimensionLabel()
        case .wideOpening:
            drawWideOpening()
        case .beam, .column:
            drawStructuralMarker()
        case .source:
            drawSourceSurface()
        }
    }

    func drawSourceSurface() {
        switch capturedSurface.category {
        case .door:
            drawDoor()
        case .wall:
            drawWall()
            updateDimensionLabel()
        case .window:
            drawWindow()
        case .opening:
            break
        @unknown default:
            break
        }
    }

    func clearPresentationShapes() {
        [selectedShape, hideWallShape, arcShape, doorSectorShape, windowGlassShape, structuralMarkerShape]
            .forEach { $0?.removeFromParent() }
        selectedShape = nil
        hideWallShape = nil
        arcShape = nil
        doorSectorShape = nil
        windowGlassShape = nil
        structuralMarkerShape = nil
    }
    
    func drawWall(with wallWidth: CGFloat = DrawParameters.wallWidth) {
        let wallPath = CGMutablePath.create(from: pointA, to: pointB)
        if let existingWall = selectedShape {
            existingWall.path = wallPath
        } else {
            let newWallShape = SKShapeNode.create(
                from: wallPath,
                strokeColor: AppTheme.wallStroke,
                lineWidth: wallWidth
            )
            newWallShape.lineCap = .square
            selectedShape = newWallShape
            addChild(newWallShape)
        }
    }
    
    func drawDoor() {
        let doorPath = CGMutablePath.create(from: pointA, to: pointC)
        let hideWallPath = CGMutablePath.create(from: pointA, to: pointB)
        let doorArcPath = createArcPath()
        let sectorPath = createDoorSectorPath()
        
        if let hideWallShape = hideWallShape {
            hideWallShape.path = hideWallPath
        } else {
            let node = SKShapeNode.create(
                from: hideWallPath,
                strokeColor: AppTheme.canvasBg,
                lineWidth: DrawParameters.hideSurfaceWidth,
                zPosition: DrawParameters.hideSurfaceZPosition
            )
            self.hideWallShape = node
            addChild(node)
        }

        if let sector = doorSectorShape {
            sector.path = sectorPath
        } else {
            let node = SKShapeNode(path: sectorPath)
            node.fillColor = AppTheme.doorFill
            node.strokeColor = .clear
            node.zPosition = DrawParameters.doorZPosition - 1
            self.doorSectorShape = node
            addChild(node)
        }
        
        if let doorShape = selectedShape {
            doorShape.path = doorPath
        } else {
            let node = SKShapeNode.create(
                from: doorPath,
                strokeColor: AppTheme.doorStroke,
                lineWidth: 8.0,
                zPosition: DrawParameters.doorZPosition
            )
            node.lineCap = .square
            selectedShape = node
            addChild(node)
        }
        
        if let doorArcShape = arcShape {
            doorArcShape.path = doorArcPath
        } else {
            let node = SKShapeNode.create(
                from: doorArcPath,
                strokeColor: AppTheme.doorStroke,
                lineWidth: DrawParameters.doorArcWidth,
                zPosition: DrawParameters.doorArcZPosition
            )
            self.arcShape = node
            addChild(node)
        }
    }

    func drawWideOpening() {
        let path = CGMutablePath()
        let openingLine = CGMutablePath.create(from: pointA, to: pointB)
        path.addPath(openingLine.copy(dashingWithPhase: 0, lengths: [10, 7]))
        let jambLength: CGFloat = 12
        for point in [pointA, pointB] {
            path.move(to: CGPoint(x: point.x, y: -jambLength / 2))
            path.addLine(to: CGPoint(x: point.x, y: jambLength / 2))
        }
        if let shape = selectedShape {
            shape.path = path
        } else {
            let shape = SKShapeNode.create(
                from: path,
                strokeColor: UIColor(red: 0.12, green: 0.55, blue: 0.48, alpha: 1),
                lineWidth: 7,
                zPosition: DrawParameters.doorZPosition
            )
            shape.lineCap = .square
            selectedShape = shape
            addChild(shape)
        }
    }

    func drawStructuralMarker() {
        let markerWidth: CGFloat
        if effectiveRepresentation == .column {
            markerWidth = min(max(halfLength * 2, 24), 40)
        } else {
            markerWidth = halfLength * 2
        }
        let markerDepth = max(depthInPoints, 16)
        let rect = CGRect(
            x: -markerWidth / 2,
            y: -markerDepth / 2,
            width: markerWidth,
            height: markerDepth
        )
        if structuralMarkerShape == nil {
            let shape = SKShapeNode(rect: rect)
            let markerColor = UIColor(red: 0.47, green: 0.31, blue: 0.72, alpha: 1)
            shape.fillColor = markerColor.withAlphaComponent(0.35)
            shape.strokeColor = markerColor
            shape.lineWidth = 2
            shape.zPosition = DrawParameters.wallWidth
            structuralMarkerShape = shape
            addChild(shape)
        } else {
            structuralMarkerShape?.path = CGPath(rect: rect, transform: nil)
        }
    }
    
    func drawWindow() {
        let hideWallPath = CGMutablePath.create(from: pointA, to: pointB)
        let framePath = CGMutablePath.create(from: pointA, to: pointB)
        
        let glassPath = CGMutablePath()
        let offset: CGFloat = 5.0
        glassPath.move(to: CGPoint(x: -halfLength, y: offset))
        glassPath.addLine(to: CGPoint(x: halfLength, y: offset))
        glassPath.move(to: CGPoint(x: -halfLength, y: -offset))
        glassPath.addLine(to: CGPoint(x: halfLength, y: -offset))
        
        if let hideWallShape = hideWallShape {
            hideWallShape.path = hideWallPath
        } else {
            let node = SKShapeNode.create(
                from: hideWallPath,
                strokeColor: AppTheme.canvasBg,
                lineWidth: DrawParameters.hideSurfaceWidth,
                zPosition: DrawParameters.hideSurfaceZPosition
            )
            self.hideWallShape = node
            addChild(node)
        }
        
        if let windowShape = selectedShape {
            windowShape.path = framePath
        } else {
            let node = SKShapeNode.create(
                from: framePath,
                strokeColor: AppTheme.windowStroke.withAlphaComponent(0.35),
                lineWidth: DrawParameters.windowWidth,
                zPosition: DrawParameters.windowZPosition
            )
            node.lineCap = .butt
            selectedShape = node
            addChild(node)
        }
        
        let isFilledGlass = isFloorToCeilingGlass
        let displayedGlassPath: CGPath
        if isFilledGlass {
            displayedGlassPath = CGPath(
                rect: CGRect(x: -halfLength, y: -8, width: halfLength * 2, height: 16),
                transform: nil
            )
        } else {
            displayedGlassPath = glassPath
        }

        if let glass = windowGlassShape {
            glass.path = displayedGlassPath
            glass.fillColor = AppTheme.windowStroke.withAlphaComponent(0.28)
            glass.strokeColor = AppTheme.windowStroke
            glass.lineWidth = isFilledGlass ? 1.5 : 3.5
        } else {
            let node = SKShapeNode(path: displayedGlassPath)
            node.strokeColor = AppTheme.windowStroke
            node.lineWidth = isFilledGlass ? 1.5 : 3.5
            node.fillColor = AppTheme.windowStroke.withAlphaComponent(0.28)
            node.zPosition = DrawParameters.windowZPosition + 1
            self.windowGlassShape = node
            addChild(node)
        }
    }
    
    func createArcPath() -> CGPath {
        let doorArcPath = CGMutablePath()
        doorArcPath.addArc(
            center: pointA,
            radius: halfLength * 2,
            startAngle: openDirection.arcStartAngle,
            endAngle: openDirection.arcEndAngle,
            clockwise: true
        )
        let dashPattern: [CGFloat] = [14.0, 8.0]
        return doorArcPath.copy(dashingWithPhase: 0, lengths: dashPattern)
    }
    
    func createDoorSectorPath() -> CGPath {
        let path = CGMutablePath()
        path.move(to: pointA)
        path.addLine(to: pointB)
        path.addArc(
            center: pointA,
            radius: halfLength * 2,
            startAngle: openDirection.arcStartAngle,
            endAngle: openDirection.arcEndAngle,
            clockwise: true
        )
        path.closeSubpath()
        return path
    }
    
    func updateHighlight() {
        let highlightRect = calculateBoundingRect()
        borderNode?.removeFromParent()
        borderNode = SKShapeNode(rect: highlightRect)
        borderNode?.strokeColor = AppTheme.windowStroke.withAlphaComponent(0.35)
        borderNode?.lineWidth = DrawParameters.wallWidth + 4
        borderNode?.zPosition = DrawParameters.wallWidth + 2
        borderNode?.glowWidth = 6
        if let borderNode = borderNode { addChild(borderNode) }
    }
    
    func calculateBoundingRect() -> CGRect {
        let width = halfLength * 2
        return CGRect(x: -width / 2, y: -2.5, width: width, height: 5.0)
    }
}

struct FloorPlanEditState: Codable {
    static let currentSchemaVersion: Int = 3
    
    static var currentAppVersion: String {
        Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "0.3.0"
    }
    
    var schemaVersion: Int
    var appVersion: String
    var createdAt: String
    var modifiedAt: String
    var displayUnit: DisplayUnit
    var calibration: SavedCalibrationState?
    
    let surfaces: [String: SavedSurfaceState]
    let objects: [String: SavedObjectState]
    let cameraRotation: Double
    var manualObjects: [SavedManualObjectState]? = nil
    var annotations: [SavedAnnotationState]? = nil
    
    enum CodingKeys: String, CodingKey {
        case schemaVersion
        case appVersion
        case createdAt
        case modifiedAt
        case displayUnit
        case calibration
        case surfaces
        case objects
        case cameraRotation
        case manualObjects
        case annotations
    }
    
    init(
        schemaVersion: Int = FloorPlanEditState.currentSchemaVersion,
        appVersion: String = FloorPlanEditState.currentAppVersion,
        createdAt: String? = nil,
        modifiedAt: String? = nil,
        displayUnit: DisplayUnit = .centimeter,
        calibration: SavedCalibrationState? = nil,
        surfaces: [String: SavedSurfaceState],
        objects: [String: SavedObjectState],
        cameraRotation: Double,
        manualObjects: [SavedManualObjectState]? = nil,
        annotations: [SavedAnnotationState]? = nil
    ) {
        let now = EditStateDateFormatter.nowString()
        self.schemaVersion = schemaVersion
        self.appVersion = appVersion
        self.createdAt = (createdAt?.isEmpty == false) ? createdAt! : now
        self.modifiedAt = (modifiedAt?.isEmpty == false) ? modifiedAt! : now
        self.displayUnit = displayUnit
        self.calibration = calibration
        self.surfaces = surfaces
        self.objects = objects
        self.cameraRotation = cameraRotation
        self.manualObjects = manualObjects
        self.annotations = annotations
    }
    
    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        self.schemaVersion = try container.decodeIfPresent(Int.self, forKey: .schemaVersion) ?? 1
        self.appVersion = try container.decodeIfPresent(String.self, forKey: .appVersion) ?? "0.2.0"
        self.createdAt = try container.decodeIfPresent(String.self, forKey: .createdAt) ?? ""
        self.modifiedAt = try container.decodeIfPresent(String.self, forKey: .modifiedAt) ?? ""
        self.displayUnit = try container.decodeIfPresent(DisplayUnit.self, forKey: .displayUnit) ?? .centimeter
        self.calibration = try container.decodeIfPresent(SavedCalibrationState.self, forKey: .calibration)
        self.surfaces = try container.decodeIfPresent([String: SavedSurfaceState].self, forKey: .surfaces) ?? [:]
        self.objects = try container.decodeIfPresent([String: SavedObjectState].self, forKey: .objects) ?? [:]
        self.cameraRotation = try container.decodeIfPresent(Double.self, forKey: .cameraRotation) ?? 0.0
        self.manualObjects = try container.decodeIfPresent([SavedManualObjectState].self, forKey: .manualObjects)
        self.annotations = try container.decodeIfPresent([SavedAnnotationState].self, forKey: .annotations)
    }
    
    static func migrateIfNeeded(
        _ state: FloorPlanEditState,
        fallbackCreatedAt: Date? = nil,
        fallbackModifiedAt: Date? = nil
    ) -> FloorPlanEditState {
        var migrated = state
        if migrated.schemaVersion < currentSchemaVersion {
            if migrated.createdAt.isEmpty {
                migrated.createdAt = EditStateDateFormatter.string(from: fallbackCreatedAt ?? Date())
            }
            if migrated.modifiedAt.isEmpty {
                migrated.modifiedAt = EditStateDateFormatter.string(from: fallbackModifiedAt ?? fallbackCreatedAt ?? Date())
            }
            migrated.schemaVersion = currentSchemaVersion
        } else if migrated.createdAt.isEmpty {
            migrated.createdAt = EditStateDateFormatter.string(from: fallbackCreatedAt ?? Date())
        }
        return migrated
    }
}

            let legacyBackupName = project.editStateFileName.replacingOccurrences(
                of: ".json",
                with: ".v\(decoded.schemaVersion).bak.json"
            )
            let legacyBackupURL = storageDirectory.appendingPathComponent(legacyBackupName)
            if FileManager.default.fileExists(atPath: url.path) &&
               !FileManager.default.fileExists(atPath: legacyBackupURL.path) {
                try? FileManager.default.copyItem(at: url, to: legacyBackupURL)
            }
            return FloorPlanEditState.migrateIfNeeded(
                decoded,
                fallbackCreatedAt: project.createdAt,
                fallbackModifiedAt: project.updatedAt
            )
        }
        return decoded
    }

        let roomURL = storageDirectory.appendingPathComponent(project.roomFileName)
        let editURL = storageDirectory.appendingPathComponent(project.editStateFileName)
        let legacyEditURL = storageDirectory.appendingPathComponent(
            project.editStateFileName.replacingOccurrences(of: ".json", with: ".v1.bak.json")
        )
        for target in [roomURL, backupURL(for: roomURL), editURL, backupURL(for: editURL), legacyEditURL] {
            try? fm.removeItem(at: target)
        }
        projects.removeAll { $0.id == project.id }
        persistIndex()
    }

    func setWallLocked(_ locked: Bool) {
        let applyUpdate = {
            if self.isWallLocked == locked { return }
            self.isWallLocked = locked
            if let pid = self.savedProjectID {
                UserDefaults.standard.set(locked, forKey: "wall_locked_\(pid.uuidString)")
            }
        }
        if Thread.isMainThread {
            applyUpdate()
        } else {
            DispatchQueue.main.async(execute: applyUpdate)
        }
    }
    
    func loadWallLockPreference(for projectID: UUID?) {
        guard let pid = projectID else { return }
        let key = "wall_locked_\(pid.uuidString)"
        let applyLoad = {
            if UserDefaults.standard.object(forKey: key) != nil {
                self.isWallLocked = UserDefaults.standard.bool(forKey: key)
            } else {
                self.isWallLocked = false
            }
        }
        if Thread.isMainThread {
            applyLoad()
        } else {
            DispatchQueue.main.async(execute: applyLoad)
        }
    }
    
    func clearDirty() {
        DispatchQueue.main.async {
            self.isDirty = false
        }
    }

            print("Error: CapturedRoom.json not found in the bundle.")
            return nil
        }
        
        // Step 2: Load the data from the JSON file
        do {
            let data = try Data(contentsOf: url)
            
            // Step 3: Decode the data into a CapturedRoom object
            let decoder = JSONDecoder()
            let capturedRoom = try decoder.decode(CapturedRoom.self, from: data)
            
            return capturedRoom
        } catch {
            print("Error decoding CapturedRoom.json: \(error.localizedDescription)")
            return nil
        }
    }
}

    public let roomCaptureView: RoomCaptureView

    private let captureSessionConfig: RoomCaptureSession.Configuration

    // The final scan result

    @Published var isShowingFloorPlan: Bool = false
    
    init() {

    // Start and stop the capture session. Available from our RoomCaptureScanView.

        // Prevent the screen from sleeping
        UIApplication.shared.isIdleTimerDisabled = true
    }
    
    func stopSession() {
        isScanning = false

    }

// MARK: RoomCaptureSessionDelegate methods
extension RoomCaptureViewModel: RoomCaptureSessionDelegate {
    func captureSession(
        _ session: RoomCaptureSession,
        didEndWith data: CapturedRoomData,
        error: Error?
    ) {
        // Step 1: Handle errors
        if let error = error {
            handleCaptureError(error)
            return
        }
        
        // Step 2: Process the captured data
        Task { [weak self] in
            guard let self else { return }
            
            do {
                let room = try await roomBuilder.capturedRoom(from: data)
                
                // Update the UI or state safely on the main thread
                await MainActor.run {
                    self.handleSuccessfulRoomCapture(room)
                }
            } catch {
                print("Error creating CapturedRoom: \(error.localizedDescription)")
                self.handleRoomProcessingError(error)
            }
        }
    }
}
